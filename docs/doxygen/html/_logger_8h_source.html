102/*
103 * MISRA deviate 20.10 / 20.12: these macros use the GNU ##__VA_ARGS__
104 * extension to allow zero-argument variadic calls, e.g. LOG_INFO("msg").
105 * This is a deliberate, well-understood deviation from strict ISO C99/C11
106 * (-Wpedantic flags it) — the same idiom used throughout the Linux kernel
107 * and FreeRTOS logging subsystems. -Wpedantic is intentionally excluded
108 * from this project's warning set (see Makefile) for this reason.
109 */
110#define LOG_TRACE(fmt, ...) \
111 Logger_Write(LOG_LEVEL_TRACE, __FILE__, (uint32_t)__LINE__, __func__, \
112 (fmt), ##__VA_ARGS__)
113
114#define LOG_DEBUG(fmt, ...) \
115 Logger_Write(LOG_LEVEL_DEBUG, __FILE__, (uint32_t)__LINE__, __func__, \
116 (fmt), ##__VA_ARGS__)
