54ECUS_INLINE uint16_t EndianUtils_Swap16(uint16_t val)
55{
56#if defined(__GNUC__) || defined(__clang__)
57 return __builtin_bswap16(val);
58#else
59 return (uint16_t)(((val & 0x00FFU) << 8U) |
60 ((val & 0xFF00U) >> 8U));
61#endif
62}

118/* =========================================================================
119 * Unaligned memory pack / unpack (big-endian wire format)
120 *
121 * CAN frame payloads are byte arrays; reading a uint16 or uint32 directly
122 * from an unaligned address is undefined behaviour. These helpers perform
123 * safe byte-by-byte assembly.
124 * ========================================================================= */

131ECUS_INLINE uint16_t EndianUtils_ReadBE16(const uint8_t * restrict buf)
132{
133 return (uint16_t)(((uint16_t)buf[0] << 8U) |
134 (uint16_t)buf[1]);
135}

142ECUS_INLINE uint32_t EndianUtils_ReadBE32(const uint8_t * restrict buf)
143{
144 return ((uint32_t)buf[0] << 24U) |
145 ((uint32_t)buf[1] << 16U) |
146 ((uint32_t)buf[2] << 8U) |
147 (uint32_t)buf[3];
148}

166ECUS_INLINE void EndianUtils_WriteBE32(uint8_t * restrict buf, uint32_t val)
167{

185ECUS_INLINE bool EndianUtils_IsHostBigEndian(void)
186{
187 /* MISRA deviate 19.2: union type-punning is intentional here for
188 * endianness detection — a well-known and portable idiom. */
189 MISRA_DEVIATE(19.2, "Union used intentionally for byte-order detection")
190
191 union { uint16_t word; uint8_t bytes[2]; } probe;
192 probe.word = 0x0102U;
193 return (probe.bytes[0] == 0x01U);
194}
