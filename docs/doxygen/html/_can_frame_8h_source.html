40/* =========================================================================
41 * CAN ID with bit-fields (union overlay)
42 *
43 * On the wire (CAN 2.0B extended):
44 * Bits [28:0] = 29-bit Identifier
45 * Bit [29] = Reserved (must be 0 for data frames)
46 * Bit [30] = RTR (Remote Transmission Request)
47 * Bit [31] = IDE (ID Extension: 0=standard, 1=extended)
48 *
49 * MISRA note: bit-field signedness is implementation-defined for types
50 * other than unsigned int; we use uint32_t members only.
51 * ========================================================================= */
52typedef union CanId
53{
54 uint32_t raw;
55
56 struct
57 {
58 uint32_t id : 29;
59 uint32_t rsvd : 1;
60 uint32_t rtr : 1;
61 uint32_t ide : 1;
62 } fields;
63} CanId;
